\[
K_cf=-f''+cf,\quad
D(K_c)=\{f\in H^2(-1,1):f'(1)=f'(-1)=(f(1)-f(-1))/2\},\quad c>0.
\]

\[
p_0=1,\quad p_1=x,\quad
p_{2m}=x^{2m}-\frac m{m-1}x^{2m-2},\quad
p_{2m+1}=x^{2m+1}-\frac m{m-1}x^{2m-1},\qquad m\ge2.
